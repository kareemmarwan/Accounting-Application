async function loadFont(file: string): Promise<ArrayBuffer> {
  const buffer = readFileSync(join(process.cwd(), "public", "fonts", file));
  return buffer.buffer.slice(
    buffer.byteOffset,
    buffer.byteOffset + buffer.byteLength
  ) as ArrayBuffer;
}
